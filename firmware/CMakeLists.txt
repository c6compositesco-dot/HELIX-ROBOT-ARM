cmake_minimum_required(VERSION 3.20)
project(helix_firmware_foundation VERSION 0.1.0 LANGUAGES CXX)

option(HELIX_BUILD_BOARD_IMAGE "Build a flashable image (not implemented)" OFF)
option(HELIX_SANITIZERS "Enable host address and undefined-behaviour sanitizers" OFF)
if(HELIX_BUILD_BOARD_IMAGE)
  message(FATAL_ERROR "Board image intentionally unavailable: qualify hardware, startup, linker, pin map and stop behaviour first.")
endif()

add_library(helix_core INTERFACE)
target_include_directories(helix_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(helix_core INTERFACE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(helix_core INTERFACE -fno-exceptions -fno-rtti -fno-fast-math)
endif()

enable_testing()
add_executable(helix_contract_tests tests/contracts.cpp)
target_link_libraries(helix_contract_tests PRIVATE helix_core)
set_target_properties(helix_contract_tests PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(helix_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(HELIX_SANITIZERS)
    target_compile_options(helix_contract_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(helix_contract_tests PRIVATE -fsanitize=address,undefined)
  endif()
elseif(HELIX_SANITIZERS)
  message(FATAL_ERROR "HELIX_SANITIZERS requires GCC or Clang")
endif()
foreach(suite transmission feedback motion runtime protocol drives kinematics tasks)
  add_test(NAME ${suite} COMMAND helix_contract_tests ${suite})
endforeach()
